import copyimport datetimeimport jsonDEFAULT_ROOMS = {    "A101": {"HALL": "LUMUMBA", "capacity": 8, "occupants": []},    "A102": {"HALL": "KAGUTA", "capacity": 5, "occupants": []},    "B101": {"HALL": "SIR ALBERT COOK", "capacity": 7, "occupants": []},    "B102": {"HALL": "NANZIRI", "capacity": 5, "occupants": []},    "C101": {"HALL": "WAMALA", "capacity": 6, "occupants": []},    "C102": {"HALL": "AKII BUA HALL", "capacity": 4, "occupants": []},    "C103": {"HALL": "MANJERI HALL", "capacity": 4, "occupants": []},    "C104": {"HALL": "INZIKURU", "capacity": 3, "occupants": []},}def allocate_room(students, rooms):    reg_no = input("Enter registration number: ").strip()    if reg_no in students:        print("A student with this registration number already exists.")        return    name = input("Enter student name: ").strip()    room_no = input("Enter room number: ").strip().upper()    if room_no not in rooms:        print("Room does not exist.")        return    room = rooms[room_no]    if len(room["occupants"]) >= room["capacity"]:        print("Allocation rejected: this room is already full.")        return    students[reg_no] = {        "name": name,        "room": room_no,        "total_fee": 1000000.0,        "amount_paid": 0.0,        "balance": 1000000.0,    }    room["occupants"].append(reg_no)    print(f"{name} has successfully been allocated to room {room_no}.")def record_payment(students, payments):    reg_no = input("Enter student registration number: ").strip()    if reg_no not in students:        print("Student not found.")        return    student = students[reg_no]    print(f"Student: {student['name']}")    print(f"Outstanding balance: UGX {student['balance']:,.2f}")    try:        amount = float(input("Enter payment amount: "))        if amount <= 0:            print("Payment must be greater than zero.")            return        if amount > student["balance"]:            print("Payment cannot be greater than the outstanding balance.")            return        student["amount_paid"] += amount        student["balance"] -= amount        payments.append(            {                "reg_no": reg_no,                "amount": amount,                "date": datetime.date.today().isoformat(),            }        )        print("Payment recorded successfully.")    except ValueError:        print("Please enter a valid numeric amount.")def search_student(students):    search = input("Enter name or registration number: ").strip().lower()    found = False    for reg_no, student in students.items():        if search == reg_no.lower() or search in student["name"].lower():            print("\nStudent Found")            print("-" * 25)            print(f"Registration: {reg_no}")            print(f"Name:         {student['name']}")            print(f"Room:         {student['room']}")            print(f"Paid:         UGX {student['amount_paid']:,.2f}")            print(f"Balance:      UGX {student['balance']:,.2f}")            found = True    if not found:        print("No matching student was found.")def occupancy_report(rooms):    print("\n========== OCCUPANCY REPORT ==========")    halls = sorted(list({room["HALL"] for room in rooms.values()}))    for hall in halls:        print(f"\n{hall}")        print("-" * 40)        for room_no, room in rooms.items():            if room["HALL"] == hall:                occupancy = len(room["occupants"])                print(f"Room {room_no}: {occupancy}/{room['capacity']} occupied")def fee_defaulters(students):    try:        threshold_input = input("Enter upper balance threshold: ").strip()        threshold = float(threshold_input.replace(",", ""))        if threshold <= 0:            print("Threshold must be greater than zero.")            return        print("\n========== FEE DEFAULTERS ==========")        print(f"{'Reg No':<12} {'Student Name':<20} {'Balance (UGX)':>15}")        print("-" * 49)        found = False        for reg_no, student in students.items():            # Safely extract and cast balance to float, stripping commas if string            raw_balance = student.get("balance", 0)            if isinstance(raw_balance, str):                balance = float(raw_balance.replace(",", ""))            else:                balance = float(raw_balance)            # Condition: Has an unpaid balance (> 0) AND is strictly above the threshold            if  balance > threshold:                print(                    f"{reg_no:<12} {student['name']:<20} {balance:>15,.2f}"                )                found = True        if not found:            print(f"No students found with a balance between UGX 0 and UGX {threshold:,.2f}.")    except ValueError:        print("Please enter a valid numeric amount.")def save_data(students, rooms, payments):    try:        with open("students.json", "w") as file:            json.dump(students, file, indent=4)        with open("rooms.json", "w") as file:            json.dump(rooms, file, indent=4)        with open("payments.json", "w") as file:            json.dump(payments, file, indent=4)        print("Data saved successfully.")    except OSError:        print("Error: Unable to save data.")def load_data():    try:        with open("students.json", "r") as file:            students = json.load(file)        with open("rooms.json", "r") as file:            rooms = json.load(file)        with open("payments.json", "r") as file:            payments = json.load(file)        return students, rooms, payments    except (FileNotFoundError, json.JSONDecodeError):        print("No valid saved data found. Starting with fresh room data.")        # FIX: Deep copy to prevent mutating DEFAULT_ROOMS reference in memory        return {}, copy.deepcopy(DEFAULT_ROOMS), []def main():    students, rooms, payments = load_data()    while True:        print(            """======================================== HOSTEL MANAGEMENT SYSTEM========================================1. Register student and allocate room2. Record fee payment3. Search student4. View occupancy report5. View fee defaulters6. Save data7. Exit"""        )        choice = input("Enter your choice: ").strip()        if choice == "1":            allocate_room(students, rooms)        elif choice == "2":            record_payment(students, payments)        elif choice == "3":            search_student(students)        elif choice == "4":            occupancy_report(rooms)        elif choice == "5":            fee_defaulters(students)        elif choice == "6":            save_data(students, rooms, payments)        elif choice == "7":            save_data(students, rooms, payments)            print("Thank you for using the system.")            break        else:            print("Invalid choice. Please select 1-7.")if __name__ == "__main__":    main()
